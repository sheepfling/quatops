cmake_minimum_required(VERSION 3.16)
project(quatops_ros_capture)
find_package(ament_cmake REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(capture_ros src/capture_ros.cpp)
target_compile_features(capture_ros PRIVATE cxx_std_20)
target_compile_options(capture_ros PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
ament_target_dependencies(capture_ros tf2 tf2_geometry_msgs geometry_msgs)
target_link_libraries(capture_ros nlohmann_json::nlohmann_json)
install(TARGETS capture_ros DESTINATION lib/${PROJECT_NAME})
ament_package()
